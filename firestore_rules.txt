rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function isAuthenticated() {
      return request.auth != null;
    }

    function isProjectParticipant(projectId) {
      return request.auth.uid in (
        exists(/databases/$(database)/documents/projects/$(projectId)) ? 
        get(/databases/$(database)/documents/projects/$(projectId)).data.participants : 
        getAfter(/databases/$(database)/documents/projects/$(projectId)).data.participants
      );
    }

    match /users/{userId} {
      allow read: if isAuthenticated();
      allow write: if isAuthenticated() && request.auth.uid == userId;
    }

    match /projects/{projectId} {
      // Participants can read a specific doc and delete
      allow get, delete: if isAuthenticated() && (request.auth.uid in resource.data.participants);
      // Any authenticated user can query (needed for roomCode lookup during join)
      allow list: if isAuthenticated();
      allow create: if isAuthenticated() && (request.auth.uid in request.resource.data.participants);
      // Participants can update (edit project details, etc.)
      allow update: if isAuthenticated() && (request.auth.uid in resource.data.participants);

      // Join room: authenticated non-participant can only add themselves to participants
      allow update: if isAuthenticated()
        && !(request.auth.uid in resource.data.participants)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['participants', 'updatedAt'])
        && request.resource.data.participants == resource.data.participants.concat([request.auth.uid]);

      match /canvases/{canvasId} {
        allow read, write: if isAuthenticated() && isProjectParticipant(projectId);

        match /sections/{sectionId} {
          allow read, write: if isAuthenticated() && isProjectParticipant(projectId);

          match /bars/{barId} {
            allow read, write: if isAuthenticated() && isProjectParticipant(projectId);
          }
        }
      }

      match /logs/{logId} {
        allow read, create: if isAuthenticated() && isProjectParticipant(projectId);
      }
    }
    match /poc_sessions/{sessionId} {
      allow read, write: if isAuthenticated();
      
      match /logs/{logId} {
        allow read, write: if isAuthenticated();
      }
    }
  }
}
